// v1.89: cloud inventory includes redelivery, not just unsaved files.
let pcPendingRows=new Map(),pcInventoryTime='';
function renderPcPending(){
 const rows=[...pcPendingRows.values()],verified=rows.filter(a=>a.pcReceiptVerified).length;
 $('pendingCount').textContent=String(rows.length);
 const detail=$('pendingDetail');if(detail)detail.textContent='PC保存・照合待ち：'+(rows.length-verified)+'件 ／ PC保存記録あり（配信・整理待ち）：'+verified+'件';
 const time=$('pendingTime');if(time)time.textContent=pcInventoryTime?'一覧確認：'+pcInventoryTime+'（処理済み件数は順次反映）':'';
}
async function pendingAssets(){
 const states=await rest('/rest/v1/rpc/sentlog_archive_v1',{method:'POST',body:JSON.stringify({p_action:'list'})});
 if(!Array.isArray(states))throw Error('案件の同期対象を確認できません。');
 const ids=new Set(states.filter(c=>c.status==='active'&&!c.checking).map(c=>c.id));
 const fields=enc('id,project_id,kind,file_name,storage_path,mime_type,byte_size,sha256,status,captured_at,metadata');
 const active=[];let after='';
 for(;;){
  const rows=await rest('/rest/v1/sentlog_assets?kind=in.(photo,drawing)&status=in.(uploaded,pc_verified)&select='+fields+'&order=id.asc&limit=400'+(after?'&id=gt.'+enc(after):''));
  if(!Array.isArray(rows))throw Error('クラウドの残件を確認できません。');
  for(const a of rows){
   if(typeof a.id!=='string'||a.id<=after)throw Error('クラウド一覧の続きを確認できません。');
   after=a.id;if(ids.has(a.project_id))active.push(a);
  }
  if(rows.length<400)break;
 }
 const receipts=new Map();
 for(let offset=0;offset<active.length;offset+=100){
  const selected=active.slice(offset,offset+100).map(a=>a.id);
  if(selected.some(id=>!/^[0-9a-f-]{36}$/i.test(id)))throw Error('ファイルの識別情報を確認できません。');
  const rows=await rest('/rest/v1/sentlog_pc_receipts?asset_id=in.('+selected.map(enc).join(',')+')&select=asset_id,sha256,byte_size');
  if(!Array.isArray(rows))throw Error('PC保存記録を確認できません。');
  for(const r of rows)receipts.set(r.asset_id,r);
 }
 pcPendingRows=new Map(active.map(a=>{const r=receipts.get(a.id);return [a.id,{...a,pcReceiptVerified:!!r&&String(r.sha256).toLowerCase()===String(a.sha256).toLowerCase()&&Number(r.byte_size)===Number(a.byte_size)}]}));
 pcInventoryTime=new Date().toLocaleTimeString('ja-JP');renderPcPending();return [...pcPendingRows.values()];
}
async function syncOnce(){
 if(busy)return;busy=true;setSync('同期中…','busy');
 let problems=0,checked=0;
 try{
  await ensureSession();if(!(await permission()))throw Error('保存先フォルダへの書き込み許可が必要です');
  const list=await pendingAssets();
  for(const a of list){
   setSync('確認中 '+(checked+1)+' / '+list.length+'件：'+(a.file_name||'ファイル'),'busy');
   try{
    await ensureSession();
    if(a.status==='uploaded'){
     await saveVerified(a);
     const current=pcPendingRows.get(a.id);if(current)current.pcReceiptVerified=true;
     renderPcPending();
    }
    await deleteVerified(a);
    pcPendingRows.delete(a.id);renderPcPending();
   }catch(e){
    problems++;log('処理保留: '+(a.file_name||a.id)+' / '+e.message);
    if(e.code==='REAUTH_REQUIRED'||e.code==='SESSION_CHANGED')throw e;
   }
   checked++;
  }
  const remaining=await pendingAssets();
  setSync(remaining.length?'未完了：クラウド残り '+remaining.length+'件'+(problems?'（今回の処理保留 '+problems+'件）':''):'同期完了：クラウドの残件はありません',remaining.length?'warn':'ok');
 }catch(e){setSync('同期は完了していません：'+e.message,'err');log('同期確認: '+e.message)}
 finally{busy=false}
}
