/* project-sync-controls-v1: select local projects, including unsent projects. */
let projectSyncControlBusy=false;
const PROJECT_SYNC_UI_WAIT_MS=30000;
const projectSyncRequests=new Map();
let projectSyncControlSerial=0;
let projectSyncControlOwner='';
// This map owns only the caller's observation. The worker and its atomic saves
// keep their original lifetime, even when the modal stops waiting for them.
function requestProjectSync(id){
  const owner=session?.user?.id,key=JSON.stringify([owner,id]);
  if(projectSyncRequests.has(key))return projectSyncRequests.get(key);
  const task=Promise.resolve().then(()=>{
    if(!owner||session?.user?.id!==owner)throw Error('ログインが変更されました。現在のログインで同期を確認してください。');
    return syncNow({force:true,projectIds:[id]});
  });
  projectSyncRequests.set(key,task);
  const release=()=>{if(projectSyncRequests.get(key)===task)projectSyncRequests.delete(key);};
  task.then(release,release);return task;
}
async function observeProjectSync(task){
  let timer;
  try{return await Promise.race([task.then(result=>({finished:true,result})),
    new Promise(resolve=>{timer=setTimeout(()=>resolve({finished:false}),PROJECT_SYNC_UI_WAIT_MS);})]);}
  finally{clearTimeout(timer);}
}
function renderProjectSyncControls(){
  const currentOwner=session?.user?.id||'';
  if(projectSyncControlOwner!==currentOwner){projectSyncControlOwner=currentOwner;projectSyncControlBusy=false;projectSyncControlSerial++;projectSyncRequests.clear();}
  const select=document.getElementById('sentlogSyncProjectSelect'),button=document.getElementById('sentlogSyncProjectNow');
  if(!select||!button)return;
  const selected=select.value;
  const projects=(workspace().projects||[]).filter(p=>{
    if(!p?.id)return false;
    const cp=[...knownCloudProjects.values()].find(row=>row.client_key===p.id);
    return cp?.status!=='archived'&&(!archive.ready||archive.includeLocal(p.id,null));
  }).sort((a,b)=>Number(b.updatedAt||0)-Number(a.updatedAt||0));
  select.replaceChildren();
  for(const p of projects){
    const option=document.createElement('option');option.value=p.id;option.textContent=p.name||'案件';select.appendChild(option);
  }
  if(projects.some(p=>p.id===selected))select.value=selected;
  else if(projects.length)select.value=projects[0].id;
  else {const option=document.createElement('option');option.value='';option.textContent='同期できる案件がありません';select.appendChild(option);}
  select.disabled=projectSyncControlBusy||!projects.length;
  button.disabled=projectSyncControlBusy||!projects.length||!navigator.onLine||!session?.user?.id;
  button.textContent=projectSyncControlBusy?'この案件を確認中…':'この案件だけ同期';
}
async function syncSelectedProject(){
  if(projectSyncControlBusy)return;
  const owner=session?.user?.id,id=document.getElementById('sentlogSyncProjectSelect')?.value;
  if(!owner){msg('ログインしてから同期してください。');return;}
  if(!navigator.onLine){msg('オンラインになると同期を再開できます。');return;}
  const project=(workspace().projects||[]).find(p=>p.id===id);
  if(!project){msg('同期する案件を選んでください。');renderProjectSyncControls();return;}
  projectSyncControlBusy=true;projectSyncControlOwner=owner;renderProjectSyncControls();
  const serial=++projectSyncControlSerial;
  const current=()=>owner===session?.user?.id&&serial===projectSyncControlSerial&&
    document.getElementById('sentlogSyncProjectSelect')?.value===id;
  msg('「'+(project.name||'案件')+'」の記録・図面・写真を確認しています。',true);
  try{
    const task=requestProjectSync(id),observed=await observeProjectSync(task);
    if(!current())return;
    if(!observed.finished){
      msg('「'+(project.name||'案件')+'」の確認に時間がかかっています。同期完了はまだ確認できていません。処理状況をご確認ください。');
      // A later result may clear this observation, but cannot clear a message
      // from a newer click, another selected project, or a different account.
      task.then(result=>{if(current()){
        msg(result?.started?'':'同期を開始できませんでした。入力中の内容を保存し、通信とログインを確認してください。');
        syncStatus.refresh();
      }},error=>{if(current())msg(error?.message||String(error));});
      syncStatus.refresh();return;
    }
    const result=observed.result;
    if(!result?.started)msg('同期を開始できませんでした。入力中の内容を保存し、通信とログインを確認してください。');
    else msg('');
    // The sync engine owns partial-failure and completion messages. Starting a
    // request alone must never be presented here as a successful transfer.
    syncStatus.refresh();
  }catch(error){if(current())msg(error?.message||String(error));}
  finally{if(serial===projectSyncControlSerial){projectSyncControlBusy=false;renderProjectSyncControls();}}
}
