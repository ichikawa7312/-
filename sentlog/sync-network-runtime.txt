// Retry only reads. A failed write may already have reached the server, so the
// next normal sync must re-read its revision instead of blindly repeating it.
async function readSyncResponse(path,opts={}){
  const read=(opts.method||'GET').toUpperCase()==='GET';
  for(let attempt=0;;attempt++){
    const controller=new AbortController(),timer=setTimeout(()=>controller.abort(),read?12000:30000);
    try{
      const response=await fetch(SUPABASE_URL+path,{...opts,signal:controller.signal,
        headers:headers({'Content-Type':'application/json',...(opts.headers||{})})});
      const text=await response.text();
      if(read&&attempt===0&&[503,504].includes(response.status)){await sleep(400);continue;}
      return {response,text};
    }catch(error){
      const network=error?.name==='TypeError'||error?.name==='AbortError';
      if(!network)throw error;
      if(read&&attempt===0&&navigator.onLine){await sleep(400);continue;}
      throw Error('通信が途中で切れました。接続を確認して「今すぐ同期」を押してください。');
    }finally{clearTimeout(timer);}
  }
}
