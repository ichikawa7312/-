// Integrated in the cloud module by build_sync_merge.py; never imported on its own.
let mergeBusy=false;
const syncConflicts=new Map();
const MERGE_CHECKPOINT_PREFIX='sentlogSyncMergeCheckpointV1:';
let mergeNotice='';
let mergeReview=null;

function normalizeMergeSnapshot(payload){
  return window.SentlogDrawingGroups?.normalizeSnapshotPayload?.(payload)||JSON.parse(JSON.stringify(payload));
}
async function readProjectAssets(projectId,fields,{availableOnly=false}={}){
  const select=[...new Set(['id',...fields.split(',')])].join(',');
  const base='/rest/v1/sentlog_assets?project_id=eq.'+enc(projectId)+
    (availableOnly?'&status=neq.storage_deleted':'')+'&select='+enc(select)+'&order=id.asc&limit=400';
  const assets=[];let after='';
  // The API caps each response. Continue even after a short page, and use the
  // immutable ID rather than offsets so concurrent removals cannot shift rows.
  for(;;){
    const rows=await rest(base+(after?'&id=gt.'+enc(after):''));
    if(!Array.isArray(rows))throw Error('図面・写真の一覧を確認できません。もう一度同期してください。');
    if(!rows.length)return assets;
    for(const row of rows){
      if(typeof row.id!=='string'||!row.id||row.id<=after)
        throw Error('図面・写真の一覧を最後まで確認できません。もう一度同期してください。');
      after=row.id;assets.push(row);
    }
  }
}
async function checkMergeFiles(cp,local,remote){
  const verifiedPdfPageCounts={};
  const remoteIds=new Set(remote.project.drawings.map(d=>d.id));
  const shared=local.drawings.filter(d=>remoteIds.has(d.meta.id));
  if(!shared.some(d=>d.state?.sourceType||d.state?.shapes?.some(s=>s.photos?.length)))return verifiedPdfPageCounts;
  const assets=await readProjectAssets(cp.id,'client_key,kind,sha256,byte_size,metadata');
  const byKey=new Map((assets||[]).map(a=>[a.client_key,a]));
  for(const d of shared){
    if(d.state.sourceType){
      const blob=await dbGet('background:'+d.meta.id),asset=byKey.get('drawing:'+d.meta.id);
      if(!(blob instanceof Blob)||!asset||blob.size!==Number(asset.byte_size)||await sha256Hex(blob)!==String(asset.sha256).toLowerCase())
        throw Error((d.meta.name||'図面')+'：図面原本の一致を確認できません。記録を残して統合を止めました。');
      const peer=remote.drawings.find(x=>x.meta.id===d.meta.id);
      const counts=[d.meta.pageCount,d.state.pageCount,peer?.meta.pageCount,peer?.state?.pageCount];
      if(d.state.sourceType==='pdf'&&peer?.state?.sourceType==='pdf'&&counts.some(n=>n!==counts[0])){
        if(!window.pdfjsLib?.getDocument)throw Error((d.meta.name||'図面')+'：PDFのページ数を確認できません。通信状態を確認してからもう一度操作してください。');
        const task=window.pdfjsLib.getDocument({data:new Uint8Array(await blob.arrayBuffer()),isEvalSupported:false});
        try{
          const pdf=await task.promise;
          if(!Number.isSafeInteger(pdf.numPages)||pdf.numPages<1)throw Error('PDFのページ数を読み取れません。');
          // Snapshot and file uploads are separate requests. If the shared
          // snapshot describes another page count, a PDF replacement may still
          // be uploading. Never reinterpret those records using an older file.
          if(peer.state.pageCount!==pdf.numPages||peer.meta.pageCount!==pdf.numPages)
            throw Error((d.meta.name||'図面')+'：共有側のページ数とPDF原本が一致しません。図面の送信が終わってからもう一度同期してください。');
          verifiedPdfPageCounts[d.meta.id]=pdf.numPages;
        }finally{await task.destroy();}
      }
    }
    for(const shape of d.state.shapes||[])for(const photo of shape.photos||[]){
      const key='photo:'+d.meta.id+':'+photo.id,asset=byKey.get(key);
      if(!asset)continue;
      const blob=await dbGet(key);
      if(blob instanceof Blob&&(blob.size!==Number(asset.byte_size)||await sha256Hex(blob)!==String(asset.sha256).toLowerCase()))
        throw Error((d.meta.name||'図面')+'：同じ写真IDの内容が異なります。写真を残して統合を止めました。');
    }
  }
  return verifiedPdfPageCounts;
}
async function tryMergeProject(cp,snap,{manual=false,review=null}={}){
  const store=window.SentlogRecords,api=window.SentlogSyncMerge;
  if(!api||!window.sentlogSyncView?.mergeCommit)return {applied:false,reason:'統合機能を読み込めません。'};
  if(!archive.canSync(cp,null)||cp.status!=='active')return {applied:false,reason:'使用中の案件だけが対象です。'};
  await store.settled();store.assertSafe();
  const wsRaw=store.getItem(WORKSPACE_KEY),ws=JSON.parse(wsRaw),lp=ws.projects.find(p=>p.id===cp.client_key);
  if(!lp||snap?.payload?.project?.id!==lp.id)return {applied:false,reason:'案件情報を確認できません。'};
  const meta=getSyncMeta(lp.id),local=projectSnapshot(lp),remote=snap.payload;
  const expectedRecords=[[WORKSPACE_KEY,wsRaw],...lp.drawings.map(d=>[DRAWING_KEY_PREFIX+d.id,store.getItem(DRAWING_KEY_PREFIX+d.id)])];
  if(!manual&&!meta?.base_snapshot)return {applied:false,reason:'前回の同期内容が未保存です。両方の記録を残す操作を選んでください。'};
  const verifiedBase=meta?.base_snapshot&&await snapshotFingerprint(meta.base_snapshot)===meta.fingerprint?meta.base_snapshot:null;
  if(!manual&&!verifiedBase)return {applied:false,reason:'前回の同期記録の照合に失敗しました。両方の記録を残す操作を選んでください。'};
  const localFingerprint=await snapshotFingerprint(local),remoteFingerprint=await snapshotFingerprint(remote);
  if(review&&(review.projectId!==lp.id||review.localFingerprint!==localFingerprint||review.remoteFingerprint!==remoteFingerprint))
    return {applied:false,reason:'確認中に記録が更新されました。「両方の記録を残して同期」から、最新の違いを確認してください。'};
  const verifiedPdfPageCounts=await checkMergeFiles(cp,local,remote);
  const result=api.merge(normalizeMergeSnapshot(local),normalizeMergeSnapshot(remote),{
    preserve:manual,base:verifiedBase?normalizeMergeSnapshot(verifiedBase):undefined,verifiedPdfPageCounts,recordChoices:review?.choices
  });
  if(!result.ok)return {applied:false,reason:result.conflicts.slice(0,3).map(c=>c.path+'：'+c.reason).join(' / '),
    review:manual&&result.conflicts.every(c=>c.kind==='record')?{projectId:lp.id,localFingerprint,remoteFingerprint,remoteUpdatedAt:snap.updated_at,conflicts:result.conflicts,choices:{}}:null};
  const incomingFingerprint=await snapshotFingerprint(remote);
  const checkpoint={format:'sentlog-sync-checkpoint',version:1,createdAt:new Date().toISOString(),
    projectId:lp.id,cloudRevision:Number(snap.revision),local,remote,...(review?{recordChoices:review.choices}:{})};
  const key=MERGE_CHECKPOINT_PREFIX+lp.id+':'+await sha256Hex(new Blob([JSON.stringify(checkpoint)]));
  // Anchor to the remote revision, but leave the merged local work dirty. A newer
  // remote revision must be compared again; it must never be overwritten blindly.
  const nextMeta={revision:Number(snap.revision),local_updated_at:-1,remote_updated_at:snap.updated_at,
    fingerprint:incomingFingerprint,fingerprint_schema:SYNC_FINGERPRINT_SCHEMA,base_snapshot:remote};
  const committed=await window.sentlogSyncView.mergeCommit(cp,{payload:result.payload},JSON.stringify(lp),{
    expectedRecords,extraWrites:[[key,JSON.stringify(checkpoint)],[SYNC_META_PREFIX+lp.id,JSON.stringify(nextMeta)]]
  });
  if(!committed?.applied)return {applied:false,waiting:true,reason:'入力・描画を終え、図面を閉じてからもう一度操作してください。'};
  return {applied:true};
}
async function mergeSelectedProject(project,review=null){
  if(mergeBusy)return;
  mergeBusy=true;mergeNotice='現在の保存と同期が終わるのを待っています。';refreshMergePanel();
  let done=false;
  try{
    await window.sentlogCloudIdle();
    await ensureSession();await archive.refresh();
    const cps=await rest('/rest/v1/sentlog_projects?client_key=eq.'+enc(project.client_key)+'&select=id,name,status,client_key');
    const cp=cps?.[0];
    if(!cp||!archive.canSync(cp,null)||cp.status!=='active')throw Error('この案件は現在の同期対象ではありません。');
    const snapshots=await rest('/rest/v1/sentlog_project_snapshots?project_id=eq.'+enc(cp.id)+'&select=revision,payload,updated_at');
    const result=await tryMergeProject(cp,snapshots?.[0],{manual:true,review});
    mergeReview=result.review?{...result.review,project}:null;
    if(!result.applied)throw Error(result.reason);
    syncConflicts.delete(cp.client_key);done=true;
    mergeNotice='統合前の記録を端末内に控えとして保存し、両方の記録をまとめました。送受信の結果は上の同期状況で確認できます。';
  }catch(e){mergeNotice=e?.message||String(e);}
  finally{mergeBusy=false;refreshMergePanel();}
  if(done)await syncNow();
}
function refreshMergePanel(){
  if(typeof document==='undefined')return;
  const host=document.getElementById('sentlogMergePanel');if(!host)return;
  host.replaceChildren();host.hidden=!syncConflicts.size&&!mergeNotice;
  if(syncConflicts.size){
    const help=document.createElement('p');
    help.textContent='片方にだけある記録も残してまとめます。削除した記録が戻る場合があります。自動番号は振り直し、手入力の番号は保持します。同じ記録の内容が食い違う場合は止めて表示します。';
    help.style.cssText='font-size:12px;line-height:1.6';host.appendChild(help);
    for(const cp of syncConflicts.values()){
      const name=document.createElement('p');name.textContent=cp.name||'案件';name.style.fontWeight='600';
      const button=document.createElement('button');button.type='button';button.textContent='両方の記録を残して同期';
      button.style.cssText='width:100%;min-height:44px;margin:4px 0 10px;background:#065f46;color:#fff';
      button.disabled=mergeBusy;button.onclick=()=>mergeSelectedProject(cp);host.append(name,button);
    }
  }
  if(mergeNotice){const p=document.createElement('p');p.textContent=mergeNotice;p.setAttribute('role','status');p.style.cssText='font-size:12px;line-height:1.6;overflow-wrap:anywhere';host.appendChild(p);}
  if(mergeReview){
    const review=mergeReview,section=document.createElement('section');section.id='sentlogRecordReview';
    const title=document.createElement('h3');title.textContent='内容が違う記録を確認';title.style.fontSize='15px';
    const help=document.createElement('p');help.textContent='下の記録ごとに、位置・寸法・説明をどちらから採用するか選んでください。片方だけにある記録と添付写真は両方残します。採用前の両方の内容も、この端末に控えを保存します。';help.style.cssText='font-size:12px;line-height:1.6';
    section.append(title,help);
    if(review.remoteUpdatedAt){const time=document.createElement('p');time.style.fontSize='12px';time.textContent='共有側の最終送信：'+new Date(review.remoteUpdatedAt).toLocaleString('ja-JP');section.append(time);}
    const keyOf=c=>window.SentlogSyncMerge.recordChoiceKey(c.drawingId,c.recordId);
    const apply=document.createElement('button');apply.type='button';apply.textContent='選んだ内容で同期';
    const update=()=>{apply.disabled=mergeBusy||review.conflicts.some(c=>!['local','remote'].includes(review.choices[keyOf(c)]));};
    const labels={points:'位置・形状',point:'位置',angle:'矢印の向き',memo:'説明',text:'文字',manualText:'寸法の表示',lineLengthM:'長さ（m）',strokeWidth:'線の太さ',unit:'作成時の表示倍率',type:'変状名',member:'部材',rotation:'文字の回転',color:'色',page:'ページ'};
    for(const c of review.conflicts){
      const card=document.createElement('fieldset');card.style.cssText='min-width:0;margin:10px 0;padding:10px;border:1px solid #cbd5e1;border-radius:8px';
      const legend=document.createElement('legend');legend.textContent=c.path+'（'+(c.local.page||c.remote.page||1)+'ページ）';legend.style.cssText='font-size:12px;overflow-wrap:anywhere';card.append(legend);
      const details=document.createElement('details'),summary=document.createElement('summary');summary.textContent='違いの詳細';details.append(summary);
      for(const key of new Set([...Object.keys(c.local),...Object.keys(c.remote)])){
        if(['photos','globalNo','typeNo','autoLabel','recordNo'].includes(key)||window.SentlogSyncMerge.equal(c.local[key],c.remote[key]))continue;
        const p=document.createElement('p');p.style.cssText='font-size:11px;white-space:pre-wrap;overflow-wrap:anywhere';
        const value=v=>v===undefined?'未設定':typeof v==='string'?v:JSON.stringify(v);
        p.textContent=(labels[key]||key)+'\nこの端末：'+value(c.local[key])+'\n共有側：'+value(c.remote[key]);details.append(p);
      }
      card.append(details);
      const select=document.createElement('select');select.setAttribute('aria-label',c.path+'の採用元');select.style.cssText='width:100%;min-height:44px;margin-top:8px';
      for(const [value,label] of [['','採用する内容を選択'],['local','この端末の変更を採用'],['remote','共有側（送信済み）の変更を採用']]){
        const option=document.createElement('option');option.value=value;option.textContent=label;select.append(option);
      }
      select.value=review.choices[keyOf(c)]||'';select.disabled=mergeBusy;
      select.onchange=()=>{review.choices[keyOf(c)]=select.value;update();};card.append(select);section.append(card);
    }
    apply.style.cssText='width:100%;min-height:44px;background:#065f46;color:white';update();apply.onclick=()=>mergeSelectedProject(review.project,review);section.append(apply);host.append(section);
  }
}

function installMergeHistory(){
  const card=document.querySelector('#sentlogSettingsModal .modal-card');
  if(!card||document.getElementById('sentlogMergeHistory'))return;
  const details=document.createElement('details');details.id='sentlogMergeHistory';details.style.marginTop='12px';
  const title=document.createElement('summary');title.textContent='同期でまとめる前の記録';
  const description=document.createElement('p');description.textContent='両端末の記録内容を控えとして端末内に保存しています。この控えには写真・図面の画像本体は含みません。';description.style.cssText='font-size:12px;line-height:1.6';
  const button=document.createElement('button');button.type='button';button.textContent='記録の控えを書き出す';
  button.onclick=()=>{
    const S=window.SentlogRecords,checkpoints=S.keys().filter(k=>k.startsWith(MERGE_CHECKPOINT_PREFIX)).map(k=>JSON.parse(S.getItem(k)));
    if(!checkpoints.length){alert('記録をまとめた履歴はまだありません。');return;}
    const url=URL.createObjectURL(new Blob([JSON.stringify({format:'sentlog-sync-checkpoints',version:1,checkpoints})],{type:'application/json'}));
    const a=document.createElement('a');a.href=url;a.download='セントログ_同期前の記録.json';document.body.appendChild(a);a.click();a.remove();setTimeout(()=>URL.revokeObjectURL(url),60000);
  };
  details.append(title,description,button);card.appendChild(details);
}
