function getPaperMm(page=state.currentPage,setting=getPageSetting(page)){
  let w=null,h=null,source='';
  if(setting.paperSize==='custom'){
    w=Number(setting.customW)||null;h=Number(setting.customH)||null;source='任意';
  }else if(setting.paperSize && setting.paperSize!=='auto' && PAPER_SIZES[setting.paperSize]){
    [w,h]=PAPER_SIZES[setting.paperSize];source=setting.paperSize;
  }else{
    const meta=state.pdfPageInfo?.[String(page)];if(meta){w=Number(meta.wMm)||null;h=Number(meta.hMm)||null;source='PDF自動';}
  }
  if(!w||!h)return null;
  const orientation=setting.orientation||'auto';
  const rot=getPageRotation(page),quarterTurn=rot===90||rot===270;
  if(orientation==='landscape' && h>w)[w,h]=[h,w];
  if(orientation==='portrait' && w>h)[w,h]=[h,w];
  if(orientation==='auto'){
    // Determine the page orientation before the user's display rotation, then rotate the physical paper dimensions with the PDF.
    let landscape=state.width>state.height;if(quarterTurn)landscape=!landscape;
    if(landscape&&h>w)[w,h]=[h,w];if(!landscape&&w>h)[w,h]=[h,w];
  }
  if(quarterTurn)[w,h]=[h,w];
  return {w,h,source};
}

function lineLengthMm(points,page=state.currentPage){
  const denom=pageScaleDenom(page);if(!denom||!points?.length)return null;let total=0;
  for(let i=1;i<points.length;i++){const a=pointPaperMm(points[i-1],page),b=pointPaperMm(points[i],page);if(!a||!b)return null;total+=Math.hypot(b.x-a.x,b.y-a.y)}
  return total*denom;
}
function polygonAreaMm2(points,page=state.currentPage){
  const denom=pageScaleDenom(page);if(!denom||!points?.length)return null;const pts=points.map(p=>pointPaperMm(p,page));if(pts.some(p=>!p))return null;let sum=0;
  for(let i=0;i<pts.length;i++){const a=pts[i],b=pts[(i+1)%pts.length];sum+=a.x*b.y-b.x*a.y}
  return Math.abs(sum)/2*denom*denom;
}

  }
  return {kind:'length',value:lineLengthMm(s.points,page)};
}
