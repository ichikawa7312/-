async function preparePhoto(file){
  if(!file||!String(file.type||'').startsWith('image/'))return file;
  try{
    const url=URL.createObjectURL(file);
    const img=await new Promise((resolve,reject)=>{const im=new Image();im.onload=()=>resolve(im);im.onerror=reject;im.src=url});
    const maxSide=1920,ratio=Math.min(1,maxSide/Math.max(img.naturalWidth||img.width,img.naturalHeight||img.height));
    const w=Math.max(1,Math.round((img.naturalWidth||img.width)*ratio)),h=Math.max(1,Math.round((img.naturalHeight||img.height)*ratio));
    const c=document.createElement('canvas');c.width=w;c.height=h;const cctx=c.getContext('2d');cctx.drawImage(img,0,0,w,h);URL.revokeObjectURL(url);
    const blob=await new Promise(resolve=>c.toBlob(resolve,'image/jpeg',0.84));return blob||file;
  }catch{return file}
}
