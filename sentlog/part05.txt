  Object.keys(localStorage).filter(k=>k.startsWith(DRAWING_KEY_PREFIX)).forEach(k=>localStorage.removeItem(k));
  await clearDBFiles();
  workspace=pack.workspace||{projects:[]};workspace.projects=workspace.projects||[];saveWorkspace();
  for(const [id,st] of Object.entries(pack.drawingStates||{}))localStorage.setItem(drawingStorageKey(id),JSON.stringify(st));
  for(const f of pack.files||[]){const blob=dataURLToBlob(f.data);const value=f.name?new File([blob],f.name,{type:f.type||blob.type,lastModified:f.lastModified||Date.now()}):blob;await putDBFile(f.key,value)}

}
